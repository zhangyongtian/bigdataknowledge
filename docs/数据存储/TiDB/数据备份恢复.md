## 数据备份

### 官网

> https://docs.pingcap.com/zh/tidb/stable/dumpling-overview

### 操作

```
tiup install dumpling
mkdir /tmp/test

create user 'dumpling'@'10.10.1.22' identified by 'dumpling';
grant all privileges on *.* to 'dumpling'@'10.10.1.22' with grant option;

tiup dumpling -u dumpling -P 4000 -p dumpling -h 10.10.1.22 -B magellan --filetype sql --compress snappy -t 8 -o /tmp/test -r 200000 -F256MiB
```

## 数据恢复

## 官网

> https://docs.pingcap.com/zh/tidb/stable/get-started-with-tidb-lightning

### 安装恢复软件

```
tiup install tidb-lightning
```

### 编写配置文件

```
[lightning]
# 日志
level = "info"
file = "tidb-lightning.log"

[tikv-importer]
# 选择使用的导入模式
backend = "local"
# 设置排序的键值对的临时存放地址，目标路径需要是一个空目录
sorted-kv-dir = "/tmp/test/tmp"

[mydumper]
# 源数据目录。
data-source-dir = "/tmp/test/test"

# 配置通配符规则，默认规则会过滤 mysql、sys、INFORMATION_SCHEMA、PERFORMANCE_SCHEMA、METRICS_SCHEMA、INSPECTION_SCHEMA 系统数据库下的所有表
# 若不配置该项，导入系统表时会出现“找不到 schema”的异常
filter = ['*.*', '!mysql.*', '!sys.*', '!INFORMATION_SCHEMA.*', '!PERFORMANCE_SCHEMA.*', '!METRICS_SCHEMA.*', '!INSPECTION_SCHEMA.*']
[tidb]
# 目标集群的信息
host = "192.168.1.156"
port = 4000
user = "root"
password = "000000"
# 表架构信息在从 TiDB 的“状态端口”获取。
status-port = 10080
# 集群 pd 的地址
pd-addr = "192.168.1.156:2379"
```

### 执行

```
#!/bin/bash
nohup tiup tidb-lightning -config tidb-lightning.toml > nohup.out &
```
