## 安装Kafka

```shell
docker run -d --name zookeeper -p 2181:2181  -t wurstmeister/zookeeper
docker run -d --name kafka --publish 9092:9092 --link zookeeper:zookeeper -e KAFKA_BROKER_ID=1  -e HOST_IP=192.168.0.100  -e KAFKA_ZOOKEEPER_CONNECT=zookeeper:2181 -e KAFKA_ADVERTISED_LISTENERS=PLAINTEXT://192.168.0.100:9092  -e KAFKA_ADVERTISED_HOST_NAME=192.168.0.100 -e KAFKA_ADVERTISED_PORT=9082  --restart=always  -t  wurstmeister/kafka
```

## docker运行mysql

### mysql5.7
```shell
sudo docker run -d \
  --name mysql-container \
  -e MYSQL_ROOT_PASSWORD=root \
  -p 3309:3306 \
  mysql:5.7
```

### mysql8.0

```shell
sudo docker run -d \
  --name=mysql85-server \
  -p 3310:3306 \
  -e MYSQL_ROOT_PASSWORD=123456 \
  -e MYSQL_ROOT_HOST=% \
  mysql:8.1.0
```

```shell
sudo docker ps
sudo docker exec -it mysql85-server /bin/bash
```

### mysql8.0第二种方法
```
mkdir -p /home/datadev/testmysql/conf

vim /home/datadev/testmysql/conf/my.cnf

[mysqld]
server-id=1
log-bin=mysql-bin
binlog-format=ROW
gtid_mode=ON
enforce_gtid_consistency=true


sudo docker run -d \
  --restart=always \
  --name=mysql85-server \
  -p 3311:3306 \
  -e MYSQL_ROOT_PASSWORD=123456 \
  -v /home/datadev/testmysql/conf:/etc/mysql/conf.d \
  -e MYSQL_ROOT_HOST=% \
  mysql:8.0.28

-- 启动mysql容器要等6分钟左右才能够连接。

SET @@GLOBAL.GTID_MODE = ON;
SET @@GLOBAL.ENFORCE_GTID_CONSISTENCY = ON;
ALTER USER 'root'@'%' IDENTIFIED BY 'root';
FLUSH PRIVILEGES;


-- 查看是否开启gtid
SHOW VARIABLES LIKE 'gtid_mode';
-- 查看已执行的GTID集合
SELECT @@GLOBAL.gtid_executed;
SHOW VARIABLES LIKE 'binlog_format';
```

## docker安装flink
```shell
export FLINK_PROPERTIES="jobmanager.rpc.address: jobmanager"
sudo docker network create flink-network

sudo docker run -d \
    --rm \
    --name=jobmanager \
    --network flink-network \
    --publish 8082:8081 \
    --env FLINK_PROPERTIES="${FLINK_PROPERTIES}" \
    flink:latest jobmanager

sudo docker run -d \
    --rm \
    --name=taskmanager \
    --network flink-network \
    --env FLINK_PROPERTIES="${FLINK_PROPERTIES}" \
    flink:latest taskmanager

sudo docker run -d \
    --rm \
    --name=taskmanager1 \
    --network flink-network \
    --env FLINK_PROPERTIES="${FLINK_PROPERTIES}" \
    flink:latest taskmanager
```
